SELECT 
  t.name AS TableName,
  c.name AS ColumnName,
  ty.name AS DataType,
  c.max_length,
  c.is_identity
FROM sys.tables t
JOIN sys.columns c ON t.object_id = c.object_id
JOIN sys.types ty ON c.user_type_id = ty.user_type_id
WHERE t.name IN ('AspNetRoles','AspNetUsers','ApplicationRoleMenus','RoleMenuPermissions','MenuItems')
ORDER BY t.name, c.column_id;

-- PermissionId IDENTITY কিনা?
SELECT COLUMNPROPERTY(OBJECT_ID('dbo.RoleMenuPermissions'),'PermissionId','IsIdentity') AS PermissionId_IsIdentity;
